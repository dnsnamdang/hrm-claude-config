     6	// Build query string từ object params (bỏ null/undefined/'' để URL sạch)
     7	function qs(params = {}) {
     8	  const parts = []
     9	  Object.keys(params).forEach((key) => {
    10	    const val = params[key]
    11	    if (val === null || val === undefined || val === '') return
    12	    if (Array.isArray(val)) {
    13	      val.forEach((v) => {
    14	        if (v !== null && v !== undefined && v !== '') {
    15	          parts.push(`${encodeURIComponent(key)}[]=${encodeURIComponent(v)}`)
    16	        }
    17	      })
    18	    } else {
    19	      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(val)}`)
    20	    }
    21	  })
    22	  return parts.length ? `?${parts.join('&')}` : ''
    23	}

   323	                    price: p.price,
   324	                    extra_price: p.extra_price,
   325	                    allocated_price: p.allocated_price,
   326	                    rebate_price: p.rebate_price,
   327	                    // Loại (1 Firm hiện giá / 2 WrService giá 0) lấy từ HEADER phiếu.
   328	                    type: data.type,
   329	                    details: (p.details || []).map((d) => ({
   330	                        key: `d_${d.product_export_request_detail_id}`,
   331	                        product_export_request_detail_id: d.product_export_request_detail_id,
   332	                        export_request_code: d.product_export_request_code,
   333	                        // Cận trên ô "SL bán" = SL đã duyệt của dòng (BE hard-validate lại tồn thực).
   334	                        max_qty: Number(d.approved_qty) || 0,
   335	                        // Mặc định bán HẾT phần đã duyệt (Ruling B2-default) — user chỉnh giảm được.
   336	                        qty: Number(d.approved_qty) || 0,
   337	                    })),
   338	                }))
   339	
   340	                this.$nextTick(() => this.markFormPristine())
   341	            } catch (error) {
   342	                console.error('Error loading borrow-sell request:', error)
   343	                this.$toasted?.global?.error?.({ message: 'Lỗi khi tải dữ liệu phiếu yêu cầu' })
   344	                this.$router.replace('/finance/borrow-sells')
   345	            } finally {
   346	                this.$nuxt.$loading.finish()
   347	            }
   348	        },
   349	
   350	        // ───────── Số lượng "SL bán" / cận nhập (cấp dòng con) ─────────
   351	        hasCoefficient(g) {
   352	            return g.unit_coefficient !== null && g.unit_coefficient !== undefined && Number(g.unit_coefficient) !== 1
   353	        },
   354	        detailInvalid(g, d) {
   355	            const q = Number(d.qty) || 0
   356	            return q < 0 || q > d.max_qty
   357	        },
   358	        onQtyInput() {
   359	            // Sửa SL -> xoá cờ lỗi field chung để người dùng không thấy thông báo cũ trong lúc gõ.
   360	            this.clearFieldError('products')
   361	        },
   362	
   363	        // ───────── Getter tiền cấp nhóm SP (khớp getter class BE — verbatim Phase 1) ─────────
   364	        groupBorrowSellQty(g) {
   365	            return g.details.reduce((a, d) => a + (Number(d.qty) || 0), 0)
   366	        },
   367	        groupDisplayPrice(g) {
   368	            return Number(g.type) === 2 ? 0 : Number(g.price) || 0
   369	        },
   370	        groupPriceAfterExtra(g) {
   371	            return (Number(g.extra_price) || 0) + this.groupDisplayPrice(g)
   372	        },
   373	        groupTotalAfterExtra(g) {
   374	            return this.groupPriceAfterExtra(g) * this.groupBorrowSellQty(g)
   375	        },
   376	        groupDiscountPrice(g) {
   377	            return Math.round(this.groupPriceAfterExtra(g) - (Number(g.allocated_price) || 0) - (Number(g.rebate_price) || 0))
   378	        },
   379	        groupTotalAllocated(g) {
   380	            return (Number(g.allocated_price) || 0) * this.groupBorrowSellQty(g)
   381	        },
   382	        groupVatCost(g) {
   383	            return (this.groupTotalAllocated(g) * (Number(g.vat_percent) || 0)) / 100
   384	        },
   385	        groupTotalAllocatedAfterVat(g) {
   386	            return this.groupTotalAllocated(g) + this.groupVatCost(g)
   387	        },
   388	        groupBorrowSellExceedsContract(g) {
   389	            return Number(g.contract_qty || 0) - Number(g.exported_qty || 0) < this.groupBorrowSellQty(g)
   390	        },
   391	
   392	        // ───────── Lưu ─────────
   393	        // Payload chỉ-SL (Ruling T8-payload) — KHÔNG gửi tiền/giá, BE tự tính.
   394	        buildPayload() {
   395	            const products = []
   396	            this.productGroups.forEach((g) => {
   397	                const details = g.details
   398	                    .filter((d) => (Number(d.qty) || 0) > 0)
   399	                    .map((d) => ({ product_export_request_detail_id: d.product_export_request_detail_id, qty: Number(d.qty) || 0 }))
   400	                if (!details.length) return
   401	                products.push({ objectable_id: g.objectable_id, objectable_type: g.objectable_type, details })
   402	            })
   403	            return { borrow_sell_request_id: Number(this.requestId), products }
   404	        },
   405	        validateBeforeSubmit() {
   406	            const errs = {}
   407	            if (this.hasGridQtyViolation) errs.products = 'Có dòng vượt số lượng cho phép, vui lòng kiểm tra lại'
   408	            else if (!this.hasQtyEntered) errs.products = 'Vui lòng nhập số lượng bán cho ít nhất 1 dòng hàng'
   409	            this.formErrors = errs
   410	            if (Object.keys(errs).length) {
   411	                this.toastFormError('Vui lòng kiểm tra lại dữ liệu nhập')
   412	                this.$nextTick(() => scrollToInputError())
   413	                return false
   414	            }
   415	            return true
   416	        },
   417	        handleSaveError(error) {
   418	            const body = error?.response?.data
   419	            const errors = body?.errors
   420	            if (!errors || typeof errors !== 'object') {
   421	                this.$toasted?.global?.error?.({ message: body?.message || 'Lập phiếu bán thất bại' })
   422	                return
   423	            }
   424	            this.formErrors = { ...errors }
   425	            const first = Object.values(errors)[0]
   426	            this.$toasted?.global?.error?.({ message: (Array.isArray(first) ? first[0] : first) || 'Vui lòng kiểm tra lại dữ liệu nhập' })
   427	            this.$nextTick(() => scrollToInputError())
   428	        },
   429	        async save() {
   430	            if (this.saving) return
   431	            if (!this.validateBeforeSubmit()) return
   432	            try {
   433	                this.saving = true
   434	                this.$nuxt.$loading.start()
   435	                const res = await storeBorrowSell(this.$store.dispatch, this.buildPayload())
   436	                const data = res?.data || res || {}
   437	                this.$toasted?.global?.success?.({ message: res?.message || 'Lập phiếu bán thành công' })
   438	                this.markFormSaved()
   439	                this.$router.push('/finance/borrow-sells/' + data.id)
   440	            } catch (error) {
   441	                console.error('Error saving borrow-sell:', error)
   442	                this.handleSaveError(error)
   443	            } finally {
   444	                this.saving = false
   445	                this.$nuxt.$loading.finish()
   446	            }
   447	        },
   448	        goBack() {
   449	            this.$router.push('/finance/borrow-sells')
   450	        },
   451	
   452	        formatNum(v) {
   453	            return Number(v || 0).toLocaleString('en-US', { maximumFractionDigits: 3 })
   454	        },
   455	        formatMoney(v) {
   456	            return Math.round(Number(v) || 0).toLocaleString('en-US')
   457	        },
   458	    },
