```js
// hrm-client/pages/finance/borrow-sells/api.js
// Lớp gọi API cho màn "Phiếu xuất bán hàng mượn" (PXBHM) — Phase 2.
// Base /api/v1/ do apiGetMethod/apiPostMethod tự thêm → chỉ truyền path từ 'finance/...'.
const BASE = 'finance/borrow-sells'

// Build query string từ object params (bỏ null/undefined/'' để URL sạch)
function qs(params = {}) {
  const parts = []
  Object.keys(params).forEach((key) => {
    const val = params[key]
    if (val === null || val === undefined || val === '') return
    if (Array.isArray(val)) {
      val.forEach((v) => {
        if (v !== null && v !== undefined && v !== '') {
          parts.push(`${encodeURIComponent(key)}[]=${encodeURIComponent(v)}`)
        }
      })
    } else {
      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(val)}`)
    }
  })
  return parts.length ? `?${parts.join('&')}` : ''
}

// Danh sách phiếu xuất bán (getBorrowSells) — params: filter + page + per_page.
export const list = (dispatch, params = {}) =>
  dispatch('apiGetMethod', `${BASE}${qs(params)}`)

// Chi tiết 1 phiếu xuất bán (getBorrowSell) — kèm khối hạch toán.
export const show = (dispatch, id) =>
  dispatch('apiGetMethod', `${BASE}/${id}`)

// Lập phiếu xuất bán (storeBorrowSell) — payload qty-only, BE tự tính tiền.
export const store = (dispatch, payload) =>
  dispatch('apiPostMethod', { url: BASE, payload })

// Dữ liệu in phiếu (getBorrowSellPrint).
export const printData = (dispatch, id) =>
  dispatch('apiGetMethod', `${BASE}/${id}/print-data`)
```

```diff
+            // type==2 (HĐ dịch vụ) -> giá niêm yết hiển thị 0, ngược lại = price (ERP `price` getter).
+            return Number(this.data.type) === 2 ? 0 : Number(row.price) || 0
+        },
+        priceAfterExtra(row) {
+            return (Number(row.extra_price) || 0) + this.displayPrice(row)
+        },
+        totalAfterExtra(row) {
+            return this.priceAfterExtra(row) * this.borrowSellQty(row)
+        },
+        discountPrice(row) {
+            return Math.round(this.priceAfterExtra(row) - (Number(row.allocated_price) || 0) - (Number(row.rebate_price) || 0))
+        },
+        totalAllocated(row) {
+            return (Number(row.allocated_price) || 0) * this.borrowSellQty(row)
+        },
+        vatCost(row) {
+            return (this.totalAllocated(row) * (Number(row.vat_percent) || 0)) / 100
+        },
+        totalAllocatedAfterVat(row) {
+            return this.totalAllocated(row) + this.vatCost(row)
+        },
+
+        formatNum(value) {
+            if (value === null || value === undefined || value === '') return '—'
+            const num = Number(String(value).replace(/,/g, ''))
+            return isFinite(num) ? num.toLocaleString('en-US', { maximumFractionDigits: 2 }) : '—'
+        },
+        // Cột tiền: làm tròn về số nguyên rồi phân tách hàng nghìn (ERP dùng number:0).
+        formatMoney(v) {
+            return Math.round(Number(v) || 0).toLocaleString('en-US')
+        },
+
+        async loadData() {
+            try {
+                this.loading = true
+                this.$nuxt?.$loading?.start()
+                const response = await show(this.$store.dispatch, this.requestId)
+                this.data = response?.data || response || {}
+            } catch (error) {
+                console.error('Error loading borrow-sell request detail:', error)
+                const status = Number(error?.response?.status)
+                if (status === 403) {
+                    this.toast('error', error?.response?.data?.message || 'Bạn không có quyền xem phiếu này')
+                } else if (status === 404) {
+                    this.toast('error', 'Phiếu yêu cầu xuất bán hàng mượn không tồn tại')
+                } else {
+                    this.toast('error', error?.response?.data?.message || 'Lỗi khi tải dữ liệu phiếu')
+                }
+            } finally {
+                this.loading = false
+                this.$nuxt?.$loading?.finish()
+            }
+        },
+
+        askDeny() {
+            if (!this.data.is_can_deny) return
+            this.$bvModal.show('deny-borrow-sell-request-detail')
+        },
+        handleDenyConfirm(comment) {
+            const value = (comment || '').trim()
+            if (!value) {
+                this.toast('error', 'Vui lòng nhập lý do từ chối')
+                return
+            }
+            this.runRequest(() => deny(this.$store.dispatch, this.requestId, value))
+        },
+        askAction(action) {
+            this.pendingAction = action
+            this.$bvModal.show('confirm-approve-borrow-sell-request-detail')
+        },
+        runAction() {
+            if (this.pendingAction === 'manager_approve') {
+                this.runRequest(() => managerApprove(this.$store.dispatch, this.requestId))
+            } else if (this.pendingAction === 'switch_board') {
+                this.runRequest(() => switchBoardOfManager(this.$store.dispatch, this.requestId))
+            } else if (this.pendingAction === 'board_approve') {
+                this.runRequest(() => boardOfManagerApprove(this.$store.dispatch, this.requestId))
+            }
+            this.pendingAction = ''
+        },
+
+        /** Gọi 1 API duyệt/từ chối -> toast + reload `show` để cập nhật status + cờ nút. */
+        async runRequest(actionFn) {
+            if (this.submitting) return
+            this.submitting = true
+            this.$nuxt?.$loading?.start()
+            try {
+                const response = await actionFn()
+                this.toast('success', response?.message || 'Thao tác thành công')
+                await this.loadData()
+            } catch (error) {
+                console.error('Error on approve/deny action:', error)
+                const status = Number(error?.response?.status)
+                const body = error?.response?.data
+                if (status === 403) {
+                    this.toast('error', body?.message || 'Không đủ quyền!')
+                    await this.loadData()
+                } else if (status === 422) {
+                    this.toast('error', body?.message || 'Dữ liệu không hợp lệ, vui lòng kiểm tra lại')
+                } else {
+                    this.toast('error', body?.message || 'Thao tác thất bại')
+                }
+            } finally {
+                this.submitting = false
+                this.$nuxt?.$loading?.finish()
+            }
+        },
+
+        printRequest() {
+            if (!this.data.is_can_view) return
+            window.open(`/finance/borrow-sell-requests/${this.requestId}/print`, '_blank')
+        },
+        openHistory() {
+            if (!this.data.is_can_view) return
+            this.showHistoryModal = true
+        },
+        goBack() {
+            this.$router.push('/finance/borrow-sell-requests')
+        },
+        goCreateSell() {
+            this.$router.push(`/finance/borrow-sells/create?request_id=${this.requestId}`)
+        },
+    },
+}
```
